```jsx
// src/components/books/AddBookForm.jsx
import React, { useState } from 'react';
import { Card, Form, Button, Row, Col, Alert, ButtonGroup, Modal } from 'react-bootstrap';

const AddBookForm = ({ onAddBook, shelves, onCreateShelf }) => {
  const [formData, setFormData] = useState({
    title: '',
    author: '',
    isbn: '',
    genre: '',
    shelf: shelves[0]?.name || ''
  });
  const [shelfMode, setShelfMode] = useState('existing'); // 'existing', 'new', 'quick'
  const [newShelfName, setNewShelfName] = useState('');
  const [showQuickShelves, setShowQuickShelves] = useState(false);
  const [submitting, setSubmitting] = useState(false);
  const [message, setMessage] = useState(null);

  // Quick shelf suggestions based on common categories
  const quickShelfSuggestions = [
    '📚 Fiction',
    '📖 Non-Fiction', 
    '💻 Programming',
    '🔬 Science',
    '📜 History',
    '🎨 Art & Design',
    '💼 Business',
    '🧠 Self-Help',
    '📰 Biography',
    '🏛️ Philosophy'
  ];

  // Update shelf when shelves change
  React.useEffect(() => {
    if (shelves.length > 0 && !formData.shelf && shelfMode === 'existing') {
      setFormData(prev => ({ ...prev, shelf: shelves[0].name }));
    }
  }, [shelves, formData.shelf, shelfMode]);

  const handleInputChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  };

  const handleShelfModeChange = (mode) => {
    setShelfMode(mode);
    setMessage(null);
    
    if (mode === 'existing' && shelves.length > 0) {
      setFormData(prev => ({ ...prev, shelf: shelves[0].name }));
    } else if (mode !== 'existing') {
      setFormData(prev => ({ ...prev, shelf: '' }));
    }
  };

  const handleQuickShelfSelect = async (shelfName) => {
    const cleanName = shelfName.replace(/^[\u{1F300}-\u{1F9FF}]\s*/u, ''); // Remove emoji
    
    // Check if shelf already exists
    const existingShelf = shelves.find(shelf => 
      shelf.name.toLowerCase() === cleanName.toLowerCase()
    );
    
    if (existingShelf) {
      setFormData(prev => ({ ...prev, shelf: existingShelf.name }));
      setShelfMode('existing');
      setShowQuickShelves(false);
    } else {
      setNewShelfName(cleanName);
      setFormData(prev => ({ ...prev, shelf: cleanName }));
      setShelfMode('new');
      setShowQuickShelves(false);
    }
  };

  const validateForm = () => {
    if (!formData.title.trim()) {
      setMessage({ type: 'warning', text: 'Title is required!' });
      return false;
    }
    
    if (shelfMode === 'existing' && !formData.shelf) {
      setMessage({ type: 'warning', text: 'Please select a shelf!' });
      return false;
    }
    
    if (shelfMode === 'new' && !newShelfName.trim()) {
      setMessage({ type: 'warning', text: 'Please enter a name for the new shelf!' });
      return false;
    }
    
    return true;
  };

  const resetForm = () => {
    setFormData({
      title: '',
      author: '',
      isbn: '',
      genre: '',
      shelf: shelves[0]?.name || ''
    });
    setShelfMode('existing');
    setNewShelfName('');
    setMessage(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage(null);
    
    if (!validateForm()) return;

    setSubmitting(true);
    
    try {
      let finalShelfName = formData.shelf;

      // Create new shelf if needed
      if (shelfMode === 'new' && newShelfName.trim()) {
        const shelfResult = await onCreateShelf(newShelfName.trim());
        if (!shelfResult.success) {
          setMessage({ type: 'danger', text: shelfResult.error || 'Failed to create shelf' });
          setSubmitting(false);
          return;
        }
        finalShelfName = newShelfName.trim();
      }

      // Add the book
      const bookData = { ...formData, shelf: finalShelfName };
      const result = await onAddBook(bookData);
      
      if (result.success) {
        setMessage({ type: 'success', text: `Book added successfully to "${finalShelfName}"!` });
        resetForm();
      } else {
        setMessage({ type: 'danger', text: result.error || 'Failed to add book' });
      }
    } catch (error) {
      setMessage({ type: 'danger', text: 'An error occurred while adding the book' });
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <>
      <Card className="mb-4 shadow-sm">
        <Card.Header className="bg-primary text-white">
          <h5 className="mb-0">
            📚 Add New Book
          </h5>
        </Card.Header>
        <Card.Body>
          {message && (
            <Alert variant={message.type} dismissible onClose={() => setMessage(null)}>
              {message.text}
            </Alert>
          )}
          
          <Form onSubmit={handleSubmit}>
            <Row className="g-3">
              {/* Title and Author */}
              <Col md={6}>
                <Form.Group>
                  <Form.Label>Title *</Form.Label>
                  <Form.Control
                    type="text"
                    name="title"
                    value={formData.title}
                    onChange={handleInputChange}
                    placeholder="Enter book title"
                    required
                  />
                </Form.Group>
              </Col>
              
              <Col md={6}>
                <Form.Group>
                  <Form.Label>Author</Form.Label>
                  <Form.Control
                    type="text"
                    name="author"
                    value={formData.author}
                    onChange={handleInputChange}
                    placeholder="Enter author name (optional)"
                  />
                </Form.Group>
              </Col>
              
              {/* ISBN and Genre */}
              <Col md={6}>
                <Form.Group>
                  <Form.Label>ISBN</Form.Label>
                  <Form.Control
                    type="text"
                    name="isbn"
                    value={formData.isbn}
                    onChange={handleInputChange}
                    placeholder="Enter ISBN (optional)"
                  />
                </Form.Group>
              </Col>
              
              <Col md={6}>
                <Form.Group>
                  <Form.Label>Genre</Form.Label>
                  <Form.Control
                    type="text"
                    name="genre"
                    value={formData.genre}
                    onChange={handleInputChange}
                    placeholder="Enter genre (optional)"
                  />
                </Form.Group>
              </Col>
              
              {/* Shelf Selection Section */}
              <Col xs={12}>
                <Form.Group>
                  <Form.Label>Shelf Selection *</Form.Label>
                  
                  {/* Shelf Mode Selection */}
                  <ButtonGroup className="w-100 mb-3" size="sm">
                    <Button
                      variant={shelfMode === 'existing' ? 'primary' : 'outline-primary'}
                      onClick={() => handleShelfModeChange('existing')}
                      disabled={shelves.length === 0}
                    >
                      📂 Existing Shelf
                    </Button>
                    <Button
                      variant={shelfMode === 'new' ? 'primary' : 'outline-primary'}
                      onClick={() => handleShelfModeChange('new')}
                    >
                      ➕ Create New
                    </Button>
                    <Button
                      variant={shelfMode === 'quick' ? 'primary' : 'outline-primary'}
                      onClick={() => setShowQuickShelves(true)}
                    >
                      ⚡ Quick Select
                    </Button>
                  </ButtonGroup>

                  {/* Existing Shelf Dropdown */}
                  {shelfMode === 'existing' && (
                    <Form.Select
                      name="shelf"
                      value={formData.shelf}
                      onChange={handleInputChange}
                      required
                    >
                      {shelves.length === 0 ? (
                        <option value="">No shelves available</option>
                      ) : (
                        <>
                          <option value="">Select a shelf</option>
                          {shelves
                            .sort((a, b) => a.name.localeCompare(b.name))
                            .map(shelf => (
                              <option key={shelf.name} value={shelf.name}>
                                📂 {shelf.name} ({shelf.bookCount} book{shelf.bookCount !== 1 ? 's' : ''})
                              </option>
                            ))}
                        </>
                      )}
                    </Form.Select>
                  )}

                  {/* New Shelf Input */}
                  {shelfMode === 'new' && (
                    <div>
                      <Form.Control
                        type="text"
                        value={newShelfName}
                        onChange={(e) => {
                          setNewShelfName(e.target.value);
                          setFormData(prev => ({ ...prev, shelf: e.target.value }));
                        }}
                        placeholder="Enter new shelf name"
                        maxLength={50}
                        required
                      />
                      <Form.Text className="text-muted">
                        A new shelf will be created with this name
                      </Form.Text>
                    </div>
                  )}

                  {/* No Shelves Warning */}
                  {shelves.length === 0 && shelfMode === 'existing' && (
                    <Alert variant="info" className="mt-2">
                      <small>
                        📝 No shelves available. Create a new one or use quick select to get started.
                      </small>
                    </Alert>
                  )}
                </Form.Group>
              </Col>
              
              {/* Submit Button */}
              <Col xs={12}>
                <div className="d-flex justify-content-between align-items-center">
                  <Button 
                    type="submit" 
                    variant="primary" 
                    size="lg"
                    disabled={submitting}
                  >
                    {submitting ? (
                      <>
                        <span className="spinner-border spinner-border-sm me-2" />
                        Adding Book...
                      </>
                    ) : (
                      <>📚 Add Book to {shelfMode === 'new' ? 'New Shelf' : 'Shelf'}</>
                    )}
                  </Button>
                  
                  <small className="text-muted">
                    {shelfMode === 'existing' && formData.shelf && (
                      <>Will be added to: <strong>📂 {formData.shelf}</strong></>
                    )}
                    {shelfMode === 'new' && newShelfName && (
                      <>Will create: <strong>📂 {newShelfName}</strong></>
                    )}
                  </small>
                </div>
              </Col>
            </Row>
          </Form>
        </Card.Body>
      </Card>

      {/* Quick Shelf Selection Modal */}
      <Modal show={showQuickShelves} onHide={() => setShowQuickShelves(false)} size="lg">
        <Modal.Header closeButton>
          <Modal.Title>⚡ Quick Shelf Selection</Modal.Title>
        </Modal.Header>
        <Modal.Body>
          <p className="text-muted mb-3">
            Select a popular shelf category or create a custom one:
          </p>
          
          <Row className="g-2">
            {quickShelfSuggestions.map((shelfName, index) => {
              const cleanName = shelfName.replace(/^[\u{1F300}-\u{1F9FF}]\s*/u, '');
              const existingShelf = shelves.find(shelf => 
                shelf.name.toLowerCase() === cleanName.toLowerCase()
              );
              
              return (
                <Col key={index} md={6} lg={4}>
                  <Button
                    variant={existingShelf ? 'outline-success' : 'outline-primary'}
                    className="w-100 text-start"
                    onClick={() => handleQuickShelfSelect(shelfName)}
                  >
                    {shelfName}
                    {existingShelf && (
                      <Badge bg="success" className="ms-2">
                        {existingShelf.bookCount}
                      </Badge>
                    )}
                  </Button>
                </Col>
              );
            })}
          </Row>
          
          <hr />
          
          <div className="text-center">
            <Button
              variant="outline-secondary"
              onClick={() => {
                setShelfMode('new');
                setShowQuickShelves(false);
              }}
            >
              ✏️ Create Custom Shelf Name
            </Button>
          </div>
        </Modal.Body>
      </Modal>
    </>
  );
};

export default AddBookForm;
```

```jsx
// src/components/shelves/ShelfSuggestions.jsx
import React from 'react';
import { Card, Button, Row, Col, Badge } from 'react-bootstrap';

const ShelfSuggestions = ({ onCreateShelf, existingShelves }) => {
  const suggestions = [
    { name: 'Currently Reading', icon: '📖', description: 'Books you are reading now' },
    { name: 'Want to Read', icon: '📚', description: 'Books on your wishlist' },
    { name: 'Favorites', icon: '⭐', description: 'Your all-time favorite books' },
    { name: 'Reference', icon: '📋', description: 'Books for reference and learning' },
    { name: 'Classics', icon: '🏛️', description: 'Classic literature and timeless works' },
    { name: 'Recent Purchases', icon: '🛒', description: 'Books you recently bought' }
  ];

  const handleCreateSuggested = async (shelfName) => {
    const result = await onCreateShelf(shelfName);
    if (result.success) {
      // You could show a toast or update UI here
    }
  };

  const availableSuggestions = suggestions.filter(suggestion => 
    !existingShelves.some(shelf => 
      shelf.name.toLowerCase() === suggestion.name.toLowerCase()
    )
  );

  if (availableSuggestions.length === 0) {
    return null;
  }

  return (
    <Card className="mb-4 shadow-sm">
      <Card.Header className="bg-light">
        <h6 className="mb-0">💡 Suggested Shelves</h6>
      </Card.Header>
      <Card.Body>
        <Row className="g-2">
          {availableSuggestions.map((suggestion, index) => (
            <Col key={index} md={6} lg={4}>
              <Button
                variant="outline-info"
                className="w-100 text-start h-100"
                onClick={() => handleCreateSuggested(suggestion.name)}
              >
                <div>
                  <div className="fw-bold">
                    {suggestion.icon} {suggestion.name}
                  </div>
                  <small className="text-muted">
                    {suggestion.description}
                  </small>
                </div>
              </Button>
            </Col>
          ))}
        </Row>
      </Card.Body>
    </Card>
  );
};

export default ShelfSuggestions;
```
