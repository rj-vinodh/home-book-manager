```jsx
// src/components/shelves/CreateShelfForm.jsx
import React, { useState } from 'react';
import { Card, Form, Button, Alert, InputGroup } from 'react-bootstrap';

const CreateShelfForm = ({ onCreateShelf, existingShelves = [] }) => {
  const [shelfName, setShelfName] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [message, setMessage] = useState(null);

  const validateShelfName = (name) => {
    const trimmedName = name.trim();
    
    if (!trimmedName) {
      return 'Shelf name is required';
    }
    
    if (trimmedName.length < 2) {
      return 'Shelf name must be at least 2 characters';
    }
    
    if (trimmedName.length > 50) {
      return 'Shelf name cannot exceed 50 characters';
    }
    
    const validPattern = /^[A-Za-z0-9\s\-_]+$/;
    if (!validPattern.test(trimmedName)) {
      return 'Only letters, numbers, spaces, hyphens, and underscores allowed';
    }
    
    const existingShelf = existingShelves.find(shelf => 
      shelf.name.toLowerCase() === trimmedName.toLowerCase()
    );
    
    if (existingShelf) {
      return `Shelf "${trimmedName}" already exists`;
    }
    
    return null;
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage(null);
    
    const validationError = validateShelfName(shelfName);
    if (validationError) {
      setMessage({ type: 'warning', text: validationError });
      return;
    }

    setSubmitting(true);
    
    try {
      const result = await onCreateShelf(shelfName.trim());
      
      if (result.success) {
        setMessage({ type: 'success', text: `Shelf "${shelfName.trim()}" created successfully!` });
        setShelfName('');
      } else {
        setMessage({ type: 'danger', text: result.error || 'Failed to create shelf' });
      }
    } catch (error) {
      setMessage({ type: 'danger', text: 'An error occurred while creating the shelf' });
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Card className="mb-4 shadow-sm">
      <Card.Header className="bg-success text-white">
        <h5 className="mb-0">
          📂 Create New Shelf
        </h5>
      </Card.Header>
      <Card.Body>
        {message && (
          <Alert 
            variant={message.type} 
            dismissible 
            onClose={() => setMessage(null)}
            className="mb-3"
          >
            {message.text}
          </Alert>
        )}
        
        <Form onSubmit={handleSubmit}>
          <Form.Group className="mb-3">
            <Form.Label>Shelf Name</Form.Label>
            <InputGroup>
              <InputGroup.Text>📂</InputGroup.Text>
              <Form.Control
                type="text"
                value={shelfName}
                onChange={(e) => setShelfName(e.target.value)}
                placeholder="Enter shelf name (e.g., History, Science Fiction)"
                maxLength={50}
                required
              />
            </InputGroup>
            <Form.Text className="text-muted">
              Only letters, numbers, spaces, hyphens, and underscores allowed (2-50 characters)
            </Form.Text>
          </Form.Group>
          
          <div className="d-flex justify-content-between align-items-center">
            <Button 
              type="submit" 
              variant="success"
              disabled={submitting || !shelfName.trim()}
            >
              {submitting ? (
                <>
                  <span className="spinner-border spinner-border-sm me-2" />
                  Creating...
                </>
              ) : (
                <>➕ Create Shelf</>
              )}
            </Button>
            
            <small className="text-muted">
              {existingShelves.length} shelf{existingShelves.length !== 1 ? 's' : ''} available
            </small>
          </div>
        </Form>
      </Card.Body>
    </Card>
  );
};

export default CreateShelfForm;
```

```jsx
// src/components/books/AddBookForm.jsx
import React, { useState } from 'react';
import { Card, Form, Button, Row, Col, Alert } from 'react-bootstrap';

const AddBookForm = ({ onAddBook, shelves }) => {
  const [formData, setFormData] = useState({
    title: '',
    author: '',
    isbn: '',
    genre: '',
    shelf: shelves[0]?.name || ''
  });
  const [submitting, setSubmitting] = useState(false);
  const [message, setMessage] = useState(null);

  // Update shelf when shelves change
  React.useEffect(() => {
    if (shelves.length > 0 && !formData.shelf) {
      setFormData(prev => ({ ...prev, shelf: shelves[0].name }));
    }
  }, [shelves, formData.shelf]);

  const handleInputChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  };

  const validateForm = () => {
    if (!formData.title.trim() || !formData.author.trim()) {
      setMessage({ type: 'warning', text: 'Title and Author are required!' });
      return false;
    }
    
    if (!formData.shelf) {
      setMessage({ type: 'warning', text: 'Please select a shelf!' });
      return false;
    }
    
    return true;
  };

  const resetForm = () => {
    setFormData({
      title: '',
      author: '',
      isbn: '',
      genre: '',
      shelf: shelves[0]?.name || ''
    });
    setMessage(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage(null);
    
    if (!validateForm()) return;

    setSubmitting(true);
    
    try {
      const result = await onAddBook(formData);
      
      if (result.success) {
        setMessage({ type: 'success', text: 'Book added successfully!' });
        resetForm();
      } else {
        setMessage({ type: 'danger', text: result.error || 'Failed to add book' });
      }
    } catch (error) {
      setMessage({ type: 'danger', text: 'An error occurred while adding the book' });
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Card className="mb-4 shadow-sm">
      <Card.Header className="bg-primary text-white">
        <h5 className="mb-0">
          📚 Add New Book
        </h5>
      </Card.Header>
      <Card.Body>
        {message && (
          <Alert variant={message.type} dismissible onClose={() => setMessage(null)}>
            {message.text}
          </Alert>
        )}
        
        <Form onSubmit={handleSubmit}>
          <Row className="g-3">
            <Col md={6}>
              <Form.Group>
                <Form.Label>Title *</Form.Label>
                <Form.Control
                  type="text"
                  name="title"
                  value={formData.title}
                  onChange={handleInputChange}
                  placeholder="Enter book title"
                  required
                />
              </Form.Group>
            </Col>
            
            <Col md={6}>
              <Form.Group>
                <Form.Label>Author *</Form.Label>
                <Form.Control
                  type="text"
                  name="author"
                  value={formData.author}
                  onChange={handleInputChange}
                  placeholder="Enter author name"
                  required
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>ISBN</Form.Label>
                <Form.Control
                  type="text"
                  name="isbn"
                  value={formData.isbn}
                  onChange={handleInputChange}
                  placeholder="Enter ISBN (optional)"
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>Genre</Form.Label>
                <Form.Control
                  type="text"
                  name="genre"
                  value={formData.genre}
                  onChange={handleInputChange}
                  placeholder="Enter genre"
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>Shelf *</Form.Label>
                <Form.Select
                  name="shelf"
                  value={formData.shelf}
                  onChange={handleInputChange}
                  required
                >
                  {shelves.length === 0 ? (
                    <option value="">No shelves available - create one first</option>
                  ) : (
                    <>
                      <option value="">Select a shelf</option>
                      {shelves.map(shelf => (
                        <option key={shelf.name} value={shelf.name}>
                          📂 {shelf.name} ({shelf.bookCount} book{shelf.bookCount !== 1 ? 's' : ''})
                        </option>
                      ))}
                    </>
                  )}
                </Form.Select>
                {shelves.length === 0 && (
                  <Form.Text className="text-muted">
                    Create a shelf first using the form above
                  </Form.Text>
                )}
              </Form.Group>
            </Col>
            
            <Col xs={12}>
              <Button 
                type="submit" 
                variant="primary" 
                size="lg"
                disabled={submitting || shelves.length === 0}
              >
                {submitting ? (
                  <>
                    <span className="spinner-border spinner-border-sm me-2" />
                    Adding Book...
                  </>
                ) : (
                  <>📚 Add Book</>
                )}
              </Button>
              
              {shelves.length === 0 && (
                <div className="mt-2">
                  <small className="text-warning">
                    ⚠️ Please create at least one shelf before adding books
                  </small>
                </div>
              )}
            </Col>
          </Row>
        </Form>
      </Card.Body>
    </Card>
  );
};
```

```jsx
// src/components/shelves/ShelfList.jsx
import React from 'react';
import { Card, ListGroup, Badge, Button } from 'react-bootstrap';

const ShelfList = ({ shelves, onDeleteShelf, loading }) => {
  const handleDeleteShelf = async (shelfName, bookCount) => {
    if (bookCount > 0) {
      alert(`Cannot delete shelf "${shelfName}" because it contains ${bookCount} book${bookCount !== 1 ? 's' : ''}. Please move the books first.`);
      return;
    }

    if (window.confirm(`Are you sure you want to delete the shelf "${shelfName}"?`)) {
      await onDeleteShelf(shelfName);
    }
  };

  if (loading) {
    return (
      <Card className="mb-4">
        <Card.Body className="text-center">
          <div className="spinner-border text-primary me-2" />
          Loading shelves...
        </Card.Body>
      </Card>
    );
  }

  return (
    <Card className="mb-4 shadow-sm">
      <Card.Header className="bg-info text-white">
        <h5 className="mb-0">
          📂 Your Shelves ({shelves.length})
        </h5>
      </Card.Header>
      <Card.Body className="p-0">
        {shelves.length === 0 ? (
          <div className="text-center py-4">
            <div className="text-muted mb-2">📂</div>
            <p className="text-muted mb-0">No shelves created yet</p>
          </div>
        ) : (
          <ListGroup variant="flush">
            {shelves.map(shelf => (
              <ListGroup.Item 
                key={shelf.name}
                className="d-flex justify-content-between align-items-center"
              >
                <div>
                  <strong>📂 {shelf.name}</strong>
                  <div className="text-muted small">
                    {shelf.bookCount} book{shelf.bookCount !== 1 ? 's' : ''}
                  </div>
                </div>
                
                <div className="d-flex align-items-center gap-2">
                  <Badge 
                    bg={shelf.bookCount > 0 ? 'primary' : 'secondary'}
                    pill
                  >
                    {shelf.bookCount}
                  </Badge>
                  
                  {shelf.bookCount === 0 && (
                    <Button
                      variant="outline-danger"
                      size="sm"
                      onClick={() => handleDeleteShelf(shelf.name, shelf.bookCount)}
                      title="Delete empty shelf"
                    >
                      🗑️
                    </Button>
                  )}
                </div>
              </ListGroup.Item>
            ))}
          </ListGroup>
        )}
      </Card.Body>
    </Card>
  );
};

export default ShelfList;
```

### **4. Updated Main App Component**

```jsx
// src/App.js
import React, { useState } from 'react';
import { Container, Row, Col, Card, Navbar, Tab, Tabs } from 'react-bootstrap';
import { useBooks } from './hooks/useBooks';
import { useShelves } from './hooks/useShelves';
import BookList from './components/books/BookList';
import AddBookForm from './components/books/AddBookForm';
import CreateShelfForm from './components/shelves/CreateShelfForm';
import ShelfList from './components/shelves/ShelfList';
import SearchBar from './components/common/SearchBar';
import 'bootstrap/dist/css/bootstrap.min.css';

function App() {
  const { books, loading, error, addBook, deleteBook } = useBooks();
  const { shelves, loading: shelvesLoading, createShelf, refreshShelves } = useShelves();
  const [filteredBooks, setFilteredBooks] = useState([]);
  const [searchQuery, setSearchQuery] = useState('');
  const [activeTab, setActiveTab] = useState('books');

  // Update filtered books when books change
  React.useEffect(() => {
    if (searchQuery) {
      const filtered = books.filter(book =>
        book.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
        book.author.toLowerCase().includes(searchQuery.toLowerCase()) ||
        book.genre?.toLowerCase().includes(searchQuery.toLowerCase())
      );
      setFilteredBooks(filtered);
    } else {
      setFilteredBooks(books);
    }
  }, [books, searchQuery]);

  const handleAddBook = async (bookData) => {
    const result = await addBook(bookData);
    if (result.success) {
      refreshShelves(); // Update shelf counts
    }
    return result;
  };

  const handleDeleteBook = async (id) => {
    const result = await deleteBook(id);
    if (result.success) {
      refreshShelves(); // Update shelf counts
    }
    return result;
  };

  const handleCreateShelf = async (name) => {
    const result = await createShelf(name);
    if (result.success) {
      // Automatically switch to books tab after creating shelf
      setTimeout(() => setActiveTab('books'), 1000);
    }
    return result;
  };

  const stats = {
    totalBooks: books.length,
    totalShelves: shelves.length,
    totalAuthors: new Set(books.map(book => book.author)).size
  };

  return (
    <div className="App">
      {/* Navigation */}
      <Navbar bg="primary" variant="dark" expand="lg" className="mb-4">
        <Container>
          <Navbar.Brand href="#home">
            📚 Book Manager
          </Navbar.Brand>
        </Container>
      </Navbar>

      <Container>
        {/* Stats Cards */}
        <Row className="mb-4">
          <Col md={4}>
            <Card className="text-center bg-primary text-white">
              <Card.Body>
                <div className="display-4 mb-2">📚</div>
                <h3>{stats.totalBooks}</h3>
                <p>Total Books</p>
              </Card.Body>
            </Card>
          </Col>
          <Col md={4}>
            <Card className="text-center bg-success text-white">
              <Card.Body>
                <div className="display-4 mb-2">📂</div>
                <h3>{stats.totalShelves}</h3>
                <p>Shelves</p>
              </Card.Body>
            </Card>
          </Col>
          <Col md={4}>
            <Card className="text-center bg-info text-white">
              <Card.Body>
                <div className="display-4 mb-2">👥</div>
                <h3>{stats.totalAuthors}</h3>
                <p>Authors</p>
              </Card.Body>
            </Card>
          </Col>
        </Row>

        {/* Main Content Tabs */}
        <Tabs
          activeKey={activeTab}
          onSelect={(k) => setActiveTab(k)}
          className="mb-4"
        >
          {/* Books Tab */}
          <Tab eventKey="books" title="📚 Books">
            {/* Add Book Form */}
            <AddBookForm 
              onAddBook={handleAddBook}
              shelves={shelves}
            />

            {/* Search Bar */}
            <Card className="mb-4">
              <Card.Body>
                <SearchBar 
                  searchQuery={searchQuery}
                  onSearchChange={setSearchQuery}
                  shelves={shelves}
                  onShelfFilter={(shelfName) => {
                    if (shelfName === 'all') {
                      setFilteredBooks(books);
                    } else {
                      setFilteredBooks(books.filter(book => book.shelf === shelfName));
                    }
                    setSearchQuery('');
                  }}
                />
              </Card.Body>
            </Card>

            {/* Books List */}
            <Card>
              <Card.Header>
                <h5 className="mb-0">
                  📖 Your Books ({filteredBooks.length})
                </h5>
              </Card.Header>
              <Card.Body>
                <BookList 
                  books={filteredBooks}
                  loading={loading}
                  error={error}
                  onDeleteBook={handleDeleteBook}
                  onEditBook={(book) => {
                    console.log('Edit book:', book);
                  }}
                />
              </Card.Body>
            </Card>
          </Tab>

          {/* Shelves Management Tab */}
          <Tab eventKey="shelves" title="📂 Manage Shelves">
            <Row>
              <Col lg={6}>
                <CreateShelfForm 
                  onCreateShelf={handleCreateShelf}
                  existingShelves={shelves}
                />
              </Col>
              <Col lg={6}>
                <ShelfList 
                  shelves={shelves}
                  loading={shelvesLoading}
                  onDeleteShelf={async (shelfName) => {
                    // TODO: Implement delete shelf functionality
                    console.log('Delete shelf:', shelfName);
                  }}
                />
              </Col>
            </Row>
          </Tab>
        </Tabs>
      </Container>
    </div>
  );
}

export default App;
```

## **🎯 Key Improvements Made**

### **✅ Separation of Concerns**

- **📚 Books Tab**: Focus only on adding/viewing books with simple shelf dropdown
- **📂 Shelves Tab**: Dedicated area for shelf creation and management


### **✅ Improved User Experience**

1. **Simple Book Addition**: Just select from existing shelves via dropdown
2. **Dedicated Shelf Creation**: Separate form with proper validation
3. **Visual Feedback**: Clear success/error messages for each action
4. **Smart Navigation**: Auto-switch to books tab after creating shelf

### **✅ Better Validation**

- **Real-time shelf name validation**
- **Clear error messages**
- **Prevents duplicate shelf names**
- **Disables book addition if no shelves exist**


### **✅ Enhanced UI/UX**

- **Tabbed interface** for better organization
- **Visual indicators** showing book counts per shelf
- **Delete protection** for shelves with books
- **Loading states** for better feedback


## **🚀 User Workflow**

1. **First Time Users**: Create shelves first in the "Manage Shelves" tab
2. **Add Books**: Use simple dropdown in the "Books" tab
3. **Manage Shelves**: View, create, and delete shelves in dedicated tab
4. **Search \& Filter**: Use existing functionality in the books tab

This approach provides a much cleaner separation between shelf management and book addition, making the interface more intuitive and organized! 🎉
