# use react js

Looking at your Book Manager application, I'll help you migrate from your current vanilla JavaScript modular approach to React.js for a more modern and maintainable frontend. Here's a comprehensive guide to rebuild your Book Manager UI using React.

## **🚀 Setting Up React for Your Book Manager**

### **Custom Hooks for Data Management**

```jsx
// src/hooks/useBooks.js
import { useState, useEffect } from 'react';
import { bookAPI } from '../services/api';

export const useBooks = () => {
  const [books, setBooks] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const loadBooks = async () => {
    try {
      setLoading(true);
      const response = await bookAPI.getBooks();
      setBooks(response.data.data || []);
      setError(null);
    } catch (err) {
      setError('Failed to load books');
      console.error('Error loading books:', err);
    } finally {
      setLoading(false);
    }
  };

  const addBook = async (bookData) => {
    try {
      const response = await bookAPI.addBook(bookData);
      if (response.data.success) {
        setBooks(prev => [...prev, response.data.data]);
        return { success: true };
      }
    } catch (err) {
      return { success: false, error: err.message };
    }
  };

  const deleteBook = async (id) => {
    try {
      await bookAPI.deleteBook(id);
      setBooks(prev => prev.filter(book => book.id !== id));
      return { success: true };
    } catch (err) {
      return { success: false, error: err.message };
    }
  };

  useEffect(() => {
    loadBooks();
  }, []);

  return {
    books,
    loading,
    error,
    addBook,
    deleteBook,
    refreshBooks: loadBooks
  };
};
```

```jsx
// src/hooks/useShelves.js
import { useState, useEffect } from 'react';
import { shelfAPI } from '../services/api';

export const useShelves = () => {
  const [shelves, setShelves] = useState([]);
  const [loading, setLoading] = useState(true);

  const loadShelves = async () => {
    try {
      setLoading(true);
      const response = await shelfAPI.getShelves();
      setShelves(response.data.data || []);
    } catch (err) {
      console.error('Error loading shelves:', err);
      // Fallback to default shelves
      setShelves([
        { name: 'Main Shelf', bookCount: 0 },
        { name: 'Programming', bookCount: 0 },
        { name: 'Fiction', bookCount: 0 }
      ]);
    } finally {
      setLoading(false);
    }
  };

  const createShelf = async (name) => {
    try {
      const response = await shelfAPI.createShelf(name);
      if (response.data.success) {
        await loadShelves(); // Refresh shelves
        return { success: true };
      }
    } catch (err) {
      return { success: false, error: err.message };
    }
  };

  useEffect(() => {
    loadShelves();
  }, []);

  return { shelves, loading, createShelf, refreshShelves: loadShelves };
};
```

```jsx
// src/components/books/AddBookForm.jsx
import React, { useState } from 'react';
import { Card, Form, Button, Row, Col, Alert } from 'react-bootstrap';

const AddBookForm = ({ onAddBook, shelves, onCreateShelf }) => {
  const [formData, setFormData] = useState({
    title: '',
    author: '',
    isbn: '',
    genre: '',
    shelf: ''
  });
  const [isCustomShelf, setIsCustomShelf] = useState(false);
  const [customShelfName, setCustomShelfName] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [message, setMessage] = useState(null);

  const handleInputChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value
    });
  };

  const handleShelfModeChange = (useCustom) => {
    setIsCustomShelf(useCustom);
    if (useCustom) {
      setFormData({ ...formData, shelf: '' });
    } else {
      setCustomShelfName('');
    }
  };

  const validateForm = () => {
    if (!formData.title.trim() || !formData.author.trim()) {
      setMessage({ type: 'warning', text: 'Title and Author are required!' });
      return false;
    }
    
    const shelfName = isCustomShelf ? customShelfName.trim() : formData.shelf;
    if (!shelfName) {
      setMessage({ type: 'warning', text: 'Please select or enter a shelf name!' });
      return false;
    }
    
    return true;
  };

  const resetForm = () => {
    setFormData({ title: '', author: '', isbn: '', genre: '', shelf: shelves[^0]?.name || '' });
    setIsCustomShelf(false);
    setCustomShelfName('');
    setMessage(null);
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage(null);
    
    if (!validateForm()) return;

    setSubmitting(true);
    
    try {
      const shelfName = isCustomShelf ? customShelfName.trim() : formData.shelf;
      
      // Create shelf if using custom mode
      if (isCustomShelf && customShelfName.trim()) {
        const shelfResult = await onCreateShelf(customShelfName.trim());
        if (!shelfResult.success) {
          setMessage({ type: 'danger', text: shelfResult.error || 'Failed to create shelf' });
          return;
        }
      }

      // Add the book
      const bookData = { ...formData, shelf: shelfName };
      const result = await onAddBook(bookData);
      
      if (result.success) {
        setMessage({ type: 'success', text: 'Book added successfully!' });
        resetForm();
      } else {
        setMessage({ type: 'danger', text: result.error || 'Failed to add book' });
      }
    } catch (error) {
      setMessage({ type: 'danger', text: 'An error occurred while adding the book' });
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Card className="mb-4 shadow-sm">
      <Card.Header className="bg-primary text-white">
        <h5 className="mb-0">
          ➕ Add New Book
        </h5>
      </Card.Header>
      <Card.Body>
        {message && (
          <Alert variant={message.type} dismissible onClose={() => setMessage(null)}>
            {message.text}
          </Alert>
        )}
        
        <Form onSubmit={handleSubmit}>
          <Row className="g-3">
            <Col md={6}>
              <Form.Group>
                <Form.Label>Title *</Form.Label>
                <Form.Control
                  type="text"
                  name="title"
                  value={formData.title}
                  onChange={handleInputChange}
                  placeholder="Enter book title"
                  required
                />
              </Form.Group>
            </Col>
            
            <Col md={6}>
              <Form.Group>
                <Form.Label>Author *</Form.Label>
                <Form.Control
                  type="text"
                  name="author"
                  value={formData.author}
                  onChange={handleInputChange}
                  placeholder="Enter author name"
                  required
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>ISBN</Form.Label>
                <Form.Control
                  type="text"
                  name="isbn"
                  value={formData.isbn}
                  onChange={handleInputChange}
                  placeholder="Enter ISBN (optional)"
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>Genre</Form.Label>
                <Form.Control
                  type="text"
                  name="genre"
                  value={formData.genre}
                  onChange={handleInputChange}
                  placeholder="Enter genre"
                />
              </Form.Group>
            </Col>
            
            <Col md={4}>
              <Form.Group>
                <Form.Label>Shelf</Form.Label>
                
                <div className="mb-2">
                  <div className="btn-group w-100" role="group">
                    <input
                      type="radio"
                      className="btn-check"
                      id="useExisting"
                      checked={!isCustomShelf}
                      onChange={() => handleShelfModeChange(false)}
                    />
                    <label className="btn btn-outline-primary btn-sm" htmlFor="useExisting">
                      📋 Select Existing
                    </label>
                    
                    <input
                      type="radio"
                      className="btn-check"
                      id="useCustom"
                      checked={isCustomShelf}
                      onChange={() => handleShelfModeChange(true)}
                    />
                    <label className="btn btn-outline-primary btn-sm" htmlFor="useCustom">
                      ➕ Create New
                    </label>
                  </div>
                </div>
                
                {isCustomShelf ? (
                  <Form.Control
                    type="text"
                    value={customShelfName}
                    onChange={(e) => setCustomShelfName(e.target.value)}
                    placeholder="Enter new shelf name"
                    maxLength={50}
                  />
                ) : (
                  <Form.Select
                    name="shelf"
                    value={formData.shelf}
                    onChange={handleInputChange}
                    required
                  >
                    <option value="">Select a shelf</option>
                    {shelves.map(shelf => (
                      <option key={shelf.name} value={shelf.name}>
                        {shelf.name} ({shelf.bookCount} books)
                      </option>
                    ))}
                  </Form.Select>
                )}
              </Form.Group>
            </Col>
            
            <Col xs={12}>
              <Button 
                type="submit" 
                variant="primary" 
                size="lg"
                disabled={submitting}
              >
                {submitting ? (
                  <>
                    <Spinner animation="border" size="sm" className="me-2" />
                    Adding...
                  </>
                ) : (
                  <>➕ Add Book</>
                )}
              </Button>
            </Col>
          </Row>
        </Form>
      </Card.Body>
    </Card>
  );
};

export default AddBookForm;
```


### **Main App Component**

```jsx
// src/App.js
import React, { useState } from 'react';
import { Container, Row, Col, Card, Navbar, Nav } from 'react-bootstrap';
import { useBooks } from './hooks/useBooks';
import { useShelves } from './hooks/useShelves';
import BookList from './components/books/BookList';
import AddBookForm from './components/books/AddBookForm';
import SearchBar from './components/common/SearchBar';
import 'bootstrap/dist/css/bootstrap.min.css';

function App() {
  const { books, loading, error, addBook, deleteBook } = useBooks();
  const { shelves, createShelf, refreshShelves } = useShelves();
  const [filteredBooks, setFilteredBooks] = useState([]);
  const [searchQuery, setSearchQuery] = useState('');

  // Update filtered books when books change
  React.useEffect(() => {
    if (searchQuery) {
      const filtered = books.filter(book =>
        book.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
        book.author.toLowerCase().includes(searchQuery.toLowerCase()) ||
        book.genre?.toLowerCase().includes(searchQuery.toLowerCase())
      );
      setFilteredBooks(filtered);
    } else {
      setFilteredBooks(books);
    }
  }, [books, searchQuery]);

  const handleAddBook = async (bookData) => {
    const result = await addBook(bookData);
    if (result.success) {
      refreshShelves(); // Update shelf counts
    }
    return result;
  };

  const handleDeleteBook = async (id) => {
    const result = await deleteBook(id);
    if (result.success) {
      refreshShelves(); // Update shelf counts
    }
    return result;
  };

  const handleCreateShelf = async (name) => {
    return await createShelf(name);
  };

  const stats = {
    totalBooks: books.length,
    totalShelves: shelves.length,
    totalAuthors: new Set(books.map(book => book.author)).size
  };

  return (
    <div className="App">
      {/* Navigation */}
      <Navbar bg="primary" variant="dark" expand="lg" className="mb-4">
        <Container>
          <Navbar.Brand href="#home">
            📚 Book Manager
          </Navbar.Brand>
        </Container>
      </Navbar>

      <Container>
        {/* Stats Cards */}
        <Row className="mb-4">
          <Col md={4}>
            <Card className="text-center bg-primary text-white">
              <Card.Body>
                <div className="display-4 mb-2">📚</div>
                <h3>{stats.totalBooks}</h3>
                <p>Total Books</p>
              </Card.Body>
            </Card>
          </Col>
          <Col md={4}>
            <Card className="text-center bg-success text-white">
              <Card.Body>
                <div className="display-4 mb-2">📂</div>
                <h3>{stats.totalShelves}</h3>
                <p>Shelves</p>
              </Card.Body>
            </Card>
          </Col>
          <Col md={4}>
            <Card className="text-center bg-info text-white">
              <Card.Body>
                <div className="display-4 mb-2">👥</div>
                <h3>{stats.totalAuthors}</h3>
                <p>Authors</p>
              </Card.Body>
            </Card>
          </Col>
        </Row>

        {/* Add Book Form */}
        <AddBookForm 
          onAddBook={handleAddBook}
          shelves={shelves}
          onCreateShelf={handleCreateShelf}
        />

        {/* Search Bar */}
        <Card className="mb-4">
          <Card.Body>
            <SearchBar 
              searchQuery={searchQuery}
              onSearchChange={setSearchQuery}
              shelves={shelves}
              onShelfFilter={(shelfName) => {
                if (shelfName === 'all') {
                  setFilteredBooks(books);
                } else {
                  setFilteredBooks(books.filter(book => book.shelf === shelfName));
                }
                setSearchQuery('');
              }}
            />
          </Card.Body>
        </Card>

        {/* Books List */}
        <Card>
          <Card.Header>
            <h5 className="mb-0">
              📖 Your Books ({filteredBooks.length})
            </h5>
          </Card.Header>
          <Card.Body>
            <BookList 
              books={filteredBooks}
              loading={loading}
              error={error}
              onDeleteBook={handleDeleteBook}
              onEditBook={(book) => {
                // TODO: Implement edit functionality
                console.log('Edit book:', book);
              }}
            />
          </Card.Body>
        </Card>
      </Container>
    </div>
  );
}

export default App;
```

## **🚀 Integration with Your Kotlin Backend**

Your existing Kotlin Ktor API endpoints will work seamlessly:

- **GET `/api/books`** - Fetched by `useBooks` hook
- **POST `/api/books`** - Called from `AddBookForm`
- **DELETE `/api/books/{id}`** - Triggered by book cards
- **GET `/api/shelves`** - Loaded by `useShelves` hook
